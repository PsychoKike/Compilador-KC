main {
    // Ejercicio 4 (repetitivo): clasificar N ventas.
    //   a) ventas de $200 o menos
    //   b) ventas mayores a $200 pero menores a $400
    //   c) ventas de $400 o mas
    // Entradas: N y despues las N ventas
    // Salidas:  para a), b) y c): cantidad de ventas y su importe total
    int n, i, cant_a, cant_b, cant_c;
    float venta, total_a, total_b, total_c;

    cin >> n;
    i = 0;
    cant_a = 0;
    cant_b = 0;
    cant_c = 0;
    total_a = 0;
    total_b = 0;
    total_c = 0;

    while i < n
        cin >> venta;
        if venta <= 200 then
            cant_a++;
            total_a = total_a + venta;
        else
            if venta < 400 then
                cant_b++;
                total_b = total_b + venta;
            else
                cant_c++;
                total_c = total_c + venta;
            end
        end
        i++;
    end

    cout << "Ventas de 200 o menos: " << cant_a;
    cout << "Importe: " << total_a;
    cout << "Ventas entre 200 y 400: " << cant_b;
    cout << "Importe: " << total_b;
    cout << "Ventas de 400 o mas: " << cant_c;
    cout << "Importe: " << total_c;
}
