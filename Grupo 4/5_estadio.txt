main {
    // Ejercicio 5 (repetitivo): venta de boletos de un estadio.
    // Datos: P1, P2, P3, P4, P5 (precios reales de cada localidad)
    //        CLAVE, CANT por cada venta ... terminan con -1, -1
    // Salidas:
    //   c) por cada venta: clave, cantidad e importe
    //   d) boletos vendidos de cada tipo (1 a 5)
    //   e) recaudacion total del estadio
    int clave, cant, b1, b2, b3, b4, b5;
    float p1, p2, p3, p4, p5, precio, importe, total;

    cin >> p1;
    cin >> p2;
    cin >> p3;
    cin >> p4;
    cin >> p5;

    b1 = 0;
    b2 = 0;
    b3 = 0;
    b4 = 0;
    b5 = 0;
    total = 0;

    cin >> clave;
    cin >> cant;
    while clave != -1
        precio = 0;
        if clave == 1 then
            precio = p1;
            b1 = b1 + cant;
        else
            if clave == 2 then
                precio = p2;
                b2 = b2 + cant;
            else
                if clave == 3 then
                    precio = p3;
                    b3 = b3 + cant;
                else
                    if clave == 4 then
                        precio = p4;
                        b4 = b4 + cant;
                    else
                        if clave == 5 then
                            precio = p5;
                            b5 = b5 + cant;
                        end
                    end
                end
            end
        end
        importe = precio * cant;
        total = total + importe;

        cout << "Clave: " << clave;
        cout << "Cantidad: " << cant;
        cout << "Importe: " << importe;

        cin >> clave;
        cin >> cant;
    end

    cout << "Boletos tipo 1: " << b1;
    cout << "Boletos tipo 2: " << b2;
    cout << "Boletos tipo 3: " << b3;
    cout << "Boletos tipo 4: " << b4;
    cout << "Boletos tipo 5: " << b5;
    cout << "Recaudacion total: " << total;
}
