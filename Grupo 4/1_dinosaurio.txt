main {
    // Ejercicio 1 (secuencial): peso de libras a kilogramos y
    // longitud de pies a metros.
    // Entradas: peso en libras, longitud en pies
    // Salidas:  peso en kg, longitud en m
    float peso_lb, largo_pie, peso_kg, largo_m;
